\# OpenCV 摄像头云台物体颜色追踪 完整版 # 适配树莓派 Picamera2 + PCA9685舵机驱动 import cv2 import numpy as np from adafruit\_servokit import ServoKit import time import threading import ctypes import inspect from picamera2 import Picamera2 import libcamera  # ====================== 硬件初始化 ====================== # PCA9685 16路舵机驱动 kit = ServoKit(channels=16) # pan水平舵机 servo0，tilt俯仰舵机 servo1 pan = 90 tilt = 90 kit.servo\[0].angle = pan kit.servo\[1].angle = tilt  # ====================== 工具函数 ====================== def bgr8\_to\_jpeg(value, quality=75):     return bytes(cv2.imencode('.jpg', value)\[1])  # 强制关闭线程工具 def \_async\_raise(tid, exctype):     tid = ctypes.c\_long(tid)     if not inspect.isclass(exctype):         exctype = type(exctype)     res = ctypes.pythonapi.PyThreadState\_SetAsyncExc(tid, ctypes.py\_object(exctype))     if res == 0:         raise ValueError("invalid thread id")     elif res != 1:         ctypes.pythonapi.PyThreadState\_SetAsyncExc(tid, None)         raise SystemError("PyThreadState\_SetAsyncExc failed")  def stop\_thread(thread):     \_async\_raise(thread.ident, SystemExit)  # ====================== 摄像头初始化 ====================== picamera = Picamera2() config = picamera.create\_preview\_configuration(     main={"format": 'RGB888', "size": (320, 240)},     raw={"format": "SRGGB12", "size": (1920, 1080)} ) # 画面翻转，和教材一致 config\["transform"] = libcamera.Transform(hflip=0, vflip=1) picamera.configure(config) picamera.start()  width = 320 height = 240  # HSV默认阈值（教程滑块初始值） hueLow = 96 hueUp = 120 hue2Low = 50 hue2Up = 0 satLow = 157 satHigh = 255 valLow = 100 valHigh = 255  # ====================== 主追踪循环 ====================== def Video\_display():     global pan, tilt     while True:         frame = picamera.capture\_array()         hsv = cv2.cvtColor(frame, cv2.COLOR\_BGR2HSV)          # 颜色上下限         lower1 = np.array(\[hueLow, satLow, valLow])         upper1 = np.array(\[hueUp, satHigh, valHigh])         lower2 = np.array(\[hue2Low, satLow, valLow])         upper2 = np.array(\[hue2Up, satHigh, valHigh])          mask1 = cv2.inRange(hsv, lower1, upper1)         mask2 = cv2.inRange(hsv, lower2, upper2)         mask\_all = cv2.add(mask1, mask2)          # 轮廓查找         contours, \_ = cv2.findContours(mask\_all, cv2.RETR\_EXTERNAL, cv2.CHAIN\_APPROX\_SIMPLE)         # 按面积从大到小排序         contours = sorted(contours, key=lambda x: cv2.contourArea(x), reverse=True)          for cnt in contours:             area = cv2.contourArea(cnt)             (x, y, w, h) = cv2.boundingRect(cnt)             if area >= 50:                 cv2.rectangle(frame, (x, y), (x + w, y + h), (255, 0, 0), 3)                 objX = x + w / 2                 objY = y + h / 2                  # 计算中心点偏差                 errorPan = objX - width / 2                 errorTilt = objY - height / 2                  # 角度调节系数                 if abs(errorPan) > 15:                     pan = pan - errorPan / 75                 if abs(errorTilt) > 15:                     tilt = tilt - errorTilt / 75                  # 角度限幅 0\~180°                 pan = np.clip(pan, 0, 180)                 tilt = np.clip(tilt, 0, 180)                  # 舵机反向赋值（教材镜像逻辑）                 kit.servo\[0].angle = 180 - pan                 kit.servo\[1].angle = 180 - tilt                 break          # 本地预览窗口         cv2.imshow("Track Window", frame)         if cv2.waitKey(1) & 0xFF == ord('q'):             break  # ====================== 程序入口 & 安全退出 ====================== if \_\_name\_\_ == "\_\_main\_\_":     work\_thread = threading.Thread(target=Video\_display)     work\_thread.setDaemon(True)     work\_thread.start()      try:         while True:             time.sleep(0.5)     except KeyboardInterrupt:         print("\n正在关闭程序...")         stop\_thread(work\_thread)         picamera.stop()         cv2.destroyAllWindows()         print("程序已安全退出")
